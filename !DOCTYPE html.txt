<script>
// === Trạng thái game ===
const MAX_HP = 100;
let playerChoice = null;
let hp = { duc: MAX_HP, vi: MAX_HP };
let isBlocking = { duc: false, vi: false };
let gameOver = false;

// === Phần tử ===
const selectScreen = document.getElementById('select-screen');
const gameScreen = document.getElementById('game-screen');
const messageBox = document.getElementById('message');

// === Chọn nhân vật ===
document.getElementById('btn-duc').addEventListener('click', () => startGame('duc'));
document.getElementById('btn-vi').addEventListener('click', () => startGame('vi'));

function startGame(who) {
    playerChoice = who;
    selectScreen.style.display = 'none';
    gameScreen.style.display = 'block';
    resetGame();
}

function resetGame() {
    hp = { duc: MAX_HP, vi: MAX_HP };
    isBlocking = { duc: false, vi: false };
    gameOver = false;
    messageBox.style.display = 'none';
    updateHP();
}

// === Cập nhật thanh máu ===
function updateHP() {
    document.getElementById('hp-fill-duc').style.width = `${(hp.duc / MAX_HP) * 100}%`;
    document.getElementById('hp-val-duc').textContent = hp.duc;
    document.getElementById('hp-fill-vi').style.width = `${(hp.vi / MAX_HP) * 100}%`;
    document.getElementById('hp-val-vi').textContent = hp.vi;
}

// === Tạo hiệu ứng trái tim ===
function showHeart(x, y, type = '❤️') {
    const heart = document.createElement('div');
    heart.className = 'heart';
    heart.textContent = type;
    heart.style.left = `${x}px`;
    heart.style.top = `${y}px`;
    document.body.appendChild(heart);
    setTimeout(() => heart.remove(), 1000);
}

// === Hiển thị thông báo ngắn gọn (thay thế alert trên di động) ===
function showToast(text, duration = 1200) {
    const toast = document.createElement('div');
    toast.style.cssText = `
        position: absolute; top: 50%; left: 50%; transform: translate(-50%, -50%);
        background: rgba(0,0,0,0.8); color: white; padding: 12px 24px; border-radius: 10px;
        font-size: 18px; font-weight: bold; z-index: 30; text-align: center;
    `;
    toast.textContent = text;
    document.body.appendChild(toast);
    setTimeout(() => toast.remove(), duration);
}

// === Tấn công ===
document.getElementById('btn-attack').addEventListener('click', () => {
    if (gameOver) return;
    
    const attacker = playerChoice;
    const defender = attacker === 'duc' ? 'vi' : 'duc';
    
    // Hiệu ứng tấn công
    document.getElementById(`player-${attacker}`).classList.add('attacking');
    setTimeout(() => {
        document.getElementById(`player-${attacker}`).classList.remove('attacking');
    }, 300);

    // Tính sát thương
    let damage = Math.floor(Math.random() * 15) + 10; // 10-24 sát thương cơ bản
    let heartType = '❤️';
    
    if (isBlocking[defender]) {
        damage = Math.floor(damage * 0.3); // Đỡ được 70% sát thương
        heartType = '💔';
        showToast(`Đỡ! -${damage} máu`);
    } else {
        // Không đỡ: có cơ hội né 20%
        if (Math.random() < 0.2) {
            damage = 0;
            heartType = '💨';
            showToast('Né thành công!');
        }
    }

    // Áp sát thương
    if (damage > 0) {
        hp[defender] = Math.max(0, hp[defender] - damage);
        document.getElementById(`player-${defender}`).classList.add('hurt');
        setTimeout(() => {
            document.getElementById(`player-${defender}`).classList.remove('hurt');
        }, 300);
    }

    // Hiện hiệu ứng
    const defEl = document.getElementById(`player-${defender}`);
    const rect = defEl.getBoundingClientRect();
    showHeart(rect.left + rect.width/2 - 15, rect.top + 10, heartType);

    updateHP();
    
    // Kẻ địch tự phản công
    setTimeout(() => enemyAttack(defender, attacker), 900);
    
    checkGameOver();
    isBlocking[defender] = false; // Tắt trạng thái đòn
});

// === Đỡ đòn ===
document.getElementById('btn-block').addEventListener('click', () => {
    if (gameOver) return;
    isBlocking[playerChoice] = true;
    document.getElementById(`player-${playerChoice}`).classList.add('blocking');
    showToast('Đang đỡ đòn!');
    setTimeout(() => {
        document.getElementById(`player-${playerChoice}`).classList.remove('blocking');
    }, 1500); // Đỡ hiệu lực 1.5 giây
});

// === Kẻ địch tấn công tự động ===
function enemyAttack(enemy, target) {
    if (gameOver || hp[target] <= 0) return;

    // 30% xác suất địch cũng đỡ
    if (Math.random() < 0.3) {
        isBlocking[enemy] = true;
        setTimeout(() => isBlocking[enemy] = false, 1500);
        return;
    }

    // Địch tấn công
    document.getElementById(`player-${enemy}`).classList.add('attacking');
    setTimeout(() => {
        document.getElementById(`player-${enemy}`).classList.remove('attacking');
    }, 300);

    let damage = Math.floor(Math.random() * 12) + 8; // 8-19 sát thương
    let heartType = '❤️';

    if (isBlocking[target]) {
        damage = Math.floor(damage * 0.3);
        heartType = '💔';
        showToast(`Bạn đỡ được! -${damage} máu`);
    } else {
        if (Math.random() < 0.2) {
            damage = 0;
            heartType = '💨';
            showToast('Bạn né được!');
        }
    }

    if (damage > 0) {
        hp[target] = Math.max(0, hp[target] - damage);
        document.getElementById(`player-${target}`).classList.add('hurt');
        setTimeout(() => {
            document.getElementById(`player-${target}`).classList.remove('hurt');
        }, 300);
    }

    const tgtEl = document.getElementById(`player-${target}`);
    const rect = tgtEl.getBoundingClientRect();
    showHeart(rect.left + rect.width/2 - 15, rect.top + 10, heartType);

    updateHP();
    checkGameOver();
}

// === Kiểm tra kết thúc ===
function checkGameOver() {
    if (hp.duc <= 0 && hp.vi <= 0) {
        endGame('🤝 Hòa!\nCả hai cùng ngã!');
    } else if (hp.duc <= 0) {
        endGame('🔵 Vi Thắng! 🎉');
    } else if (hp.vi <= 0) {
        endGame('🔴 Đức Thắng! 🎉');
    }
}

function endGame(text) {
    gameOver = true;
    document.getElementById('result-text').innerHTML = text.replace(/\n/g, '<br>');
    messageBox.style.display = 'block';
}

document.getElementById('restart-btn').addEventListener('click', () => {
    resetGame();
    selectScreen.style.display = 'flex';
    gameScreen.style.display = 'none';
});
</script>
